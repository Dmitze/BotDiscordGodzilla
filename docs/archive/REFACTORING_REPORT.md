```javascript
function hasRole(member, requiredRoles) {
  if (!member || !member.roles) {
    return false;
  }
  const userRoles = member.roles.cache.map(role => role.name);
  return userRoles.includes(requiredRoles);
}

// Після рефакторингу
function hasRole(member, requiredRoles) {
  try {
    if (!member || !member.roles) {
      logger.warn('Invalid member object provided to hasRole');
      return false;
    }
    const userRoles = member.roles.cache.map(role => role.name);
    if (Array.isArray(requiredRoles)) {
      return requiredRoles.some(role => userRoles.includes(role));
    }
    return userRoles.includes(requiredRoles);
  } catch (error) {
    logger.error('Error in hasRole function:', error);
    return false;
  }
}
```

#### 🔧 Покращення:

```javascript
// До рефакторингу
async searchFiles(query, folderId = null) {
  let searchQuery = `name contains '${query}'`;
  const response = await this.drive.files.list({
    q: searchQuery,
    fields: 'files(id,name,mimeType,size,modifiedTime,webViewLink)',
    pageSize: 20,
  });
  return response.data.files || [];
}

// Після рефакторингу
async searchFiles(query, folderId = null) {
  try {
    if (!this.drive) {
      throw new Error('Google Drive API not initialized');
    }

    let searchQuery = `name contains '${this.sanitizeQuery(query)}'`;
    const response = await this.drive.files.list({
      q: searchQuery,
      fields: 'files(id,name,mimeType,size,modifiedTime,webViewLink)',
      pageSize: 20,
    });

    const files = response.data.files || [];
    this.stats.filesProcessed += files.length;

    logger.info(`Found ${files.length} files for query: ${query}`);
    return files;
  } catch (error) {
    this.stats.errors++;
    logger.error('File search error:', error);
    throw new Error('Помилка пошуку файлів');
  }
}
```
